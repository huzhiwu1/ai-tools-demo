# 一、为什么需要中文分词？

# --------------------------------------------------

# ES 默认的 standard 分词器只按空格和标点拆分，

# 中文句子没有空格分隔，所以 standard 会把整个中文句子当成一个 token，

# 导致中文搜索几乎无法命中！

#

# IK 分词器是专为中文设计的插件：

# ik_max_word：最细粒度切分（建索引时用，尽量多分词，提高召回率）

# ik_smart：智能切分（搜索时用，分词更自然，提高准确率）

# --------------------------------------------------

# 二、检查 IK 分词插件是否安装

# --------------------------------------------------

# 1. 查看 ES 状态

GET /

# 2. 查看已安装插件（应看到 analysis-ik）

GET /\_cat/plugins?v

# --------------------------------------------------

# 三、分词效果对比

# --------------------------------------------------

# 3. 原生 standard 分词（❌ 中文场景不适用）

# 结果：整句中文被当成一个 token

POST /\_analyze
{
"analyzer": "standard",
"text": "Elasticsearch RAG 混合检索知识库"
}

# 4. IK 细粒度分词 ik_max_word（✅ 入库存储用）

# 结果：把中文尽可能多地切分成词语

# 例如"混合检索"会被切分为"混合"、"混"、"合检"、"检索"等

POST /\_analyze
{
"analyzer": "ik_max_word",
"text": "Elasticsearch RAG 混合检索知识库"
}

# 5. IK 智能分词 ik_smart（✅ 搜索查询用）

# 结果：按照最合理的方式切分

# 例如"混合检索"只切分为"混合"、"检索"

POST /\_analyze
{
"analyzer": "ik_smart",
"text": "Elasticsearch RAG 混合检索知识库"
}

# 💡 小结：

# 入库时用 ik_max_word（多分词 → 召回更多可能）

# 搜索时用 ik_smart（少分词 → 结果更精准）

# 四、IK 双分词索引实战（生活笔记场景）

# --------------------------------------------------

# 6. 查看当前所有索引

GET /\_cat/indices?v&h=health,status,index,docs.count

# 7. 创建带 IK 分词的索引（life_note 生活笔记）

# text 字段同时指定：

# analyzer: ik_max_word（入库时细粒度分词）

# search_analyzer: ik_smart（搜索时智能分词）

# keyword 字段不分词，用于精确匹配和聚合

PUT /life_note
{
"mappings": {
"properties": {
"title": {
"type": "text",
"analyzer": "ik_max_word",
"search_analyzer": "ik_smart"
},
"content": {
"type": "text",
"analyzer": "ik_max_word",
"search_analyzer": "ik_smart"
},
"type": {
"type": "keyword"
},
"author": {
"type": "keyword"
},
"record_time": {
"type": "date"
}
}
}
}

# 8. 查看索引映射（确认分词器配置正确）

GET /life_note/\_mapping

# 9. 查看索引配置

GET /life_note/\_settings

# 六、验证 IK 分词检索效果

# --------------------------------------------------

# 12. 全文分词检索（IK 智能分词搜索）

# 搜索词"健康 作息 旅行"会被 ik_smart 切分为多个 token

# 只要文档包含其中任一 token 就会命中

GET /life_note/\_search
{
"query": {
"match": {
"content": "健康 作息 旅行"
}
}
}

# 13. 精确匹配查询（keyword 字段）

# type 是 keyword 类型，不会分词，必须完全匹配

GET /life_note/\_search
{
"query": {
"term": {
"type": "健康生活"
}
}
}

# 14. 只返回指定字段

GET /life_note/\_search
{
"\_source": ["title", "type", "author"],
"query": {
"match_all": {}
}
}

# 15. 分页 + 时间排序

GET /life_note/\_search
{
"from": 0,
"size": 10,
"sort": [
{ "record_time": "desc" }
],
"query": {
"match_all": {}
}
}
